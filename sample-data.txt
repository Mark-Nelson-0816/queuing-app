INSERT INTO players (
    name,
    level,
    gender,
    contact_number,
    prefer_mens,
    prefer_womens,
    prefer_mixed,
    prefer_no_gender,
    rank_match_preference
)
SELECT
    new_players.name,
    new_players.level,
    new_players.gender,
    'N/A',
    new_players.prefer_mens,
    new_players.prefer_womens,
    new_players.prefer_mixed,
    new_players.prefer_no_gender,
    'same_rank'
FROM new_players
WHERE NOT EXISTS (
    SELECT 1
    FROM players
    WHERE LOWER(TRIM(players.name)) = LOWER(TRIM(new_players.name))
);
